Panduan Skrip SQL Database Supabase

Jalankan di Supabase SQL Editor agar tabel talent terbuat secara otomatis

Langkah Menjalankan Skrip di Supabase:

  1. Buka Dashboard Supabase project Anda (supabase.com/dashboard).
  2. Buka menu SQL Editor di bilah navigasi kiri.
  3. Klik New query, lalu tempel (paste) skrip di bawah ini dan klik tombol Run.
-- 1. Buat Tabel Freelancers (Katalog Talent Dinamis)
CREATE TABLE IF NOT EXISTS public.freelancers (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    location TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'Available',
    service TEXT NOT NULL,
    description TEXT,
    images JSONB DEFAULT '[]'::jsonb,
    created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Aktifkan Row Level Security (RLS)
ALTER TABLE public.freelancers ENABLE ROW LEVEL SECURITY;

-- 3. Kebijakan Akses Publik (Read, Insert, Update, Delete)
CREATE POLICY "Allow public read freelancers" ON public.freelancers FOR SELECT USING (true);
CREATE POLICY "Allow public insert freelancers" ON public.freelancers FOR INSERT WITH CHECK (true);
CREATE POLICY "Allow public update freelancers" ON public.freelancers FOR UPDATE USING (true);
CREATE POLICY "Allow public delete freelancers" ON public.freelancers FOR DELETE USING (true);

-- 4. Tabel Riwayat Klik Hubungi WA Admin
CREATE TABLE IF NOT EXISTS public.wa_leads (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    member_name TEXT NOT NULL,
    talent_name TEXT NOT NULL,
    service TEXT,
    reference TEXT,
    created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);
ALTER TABLE public.wa_leads ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public all wa_leads" ON public.wa_leads FOR ALL USING (true);

-- 5. Tabel Konfigurasi WA Admin
CREATE TABLE IF NOT EXISTS public.admin_config (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    wa_number TEXT NOT NULL DEFAULT '6281234567890',
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);
ALTER TABLE public.admin_config ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public all admin_config" ON public.admin_config FOR ALL USING (true);
INSERT INTO public.admin_config (wa_number) VALUES ('6281234567890') ON CONFLICT DO NOTHING;
